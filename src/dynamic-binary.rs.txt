// Direct portable frames for the ordinary C value runtime. No document carrier.
fn binary_dynamic_value(view:value_wire::Validated<'_>)->R<V>{
 let t=view.descriptor()?;
 Ok(match view.kind()?{
  "unit"=>V::Unit,
  "bool"=>V::Bool(view.body()?[0]!=0),
  "u32"=>V::U32(u32::from_le_bytes(view.body()?.try_into().unwrap())),
  "u64"=>V::U64(u64::from_le_bytes(view.body()?.try_into().unwrap())),
  "i32"=>V::I32(i32::from_le_bytes(view.body()?.try_into().unwrap())),
  "f32"=>V::F32(u32::from_le_bytes(view.body()?.try_into().unwrap())),
  "int"=>{let text=view.text()?;if text.len()>1000000{return Err("Int input exceeds resource limit".into());}V::Int(text.parse().map_err(|_|"binary Int")?)},
  "string"=>V::String(view.text()?.into()),
  "enum"=>V::Enum(view.text()?.into()),
  "id"=>V::Id(t["name"].as_str().unwrap().into(),u128::from_str_radix(view.text()?,16).map_err(|_|"binary ID")?),
  "record"=>V::Record(t["name"].as_str().unwrap().into(),t["fields"].as_array().unwrap().iter().zip(view.fields()?).map(|(f,x)|Ok((f["name"].as_str().unwrap().into(),binary_dynamic_value(x?)?))).collect::<R<_>>()?),
  "list"=>V::List(view.items()?.map(|x|binary_dynamic_value(x?)).collect::<R<_>>()?),
  "vector"=>V::Vector(view.items()?.map(|x|binary_dynamic_value(x?)).collect::<R<_>>()?),
  "option"=>{let(_,child)=view.variant()?;V::Option(child.map(|x|binary_dynamic_value(x).map(Box::new)).transpose()?)},
  "result"=>{let(err,child)=view.variant()?;V::Result(!err,Box::new(binary_dynamic_value(child.ok_or("binary result payload")?)?))},
  _=>return Err("unsupported binary value".into()),
 })
}
fn binary_dynamic_write(s:&value_wire::Schema,t:&Json,v:&V,w:&mut value_wire::Encoder)->R<()>{
 let t=s.resolve(t)?;
 w.frame(|w|{
  match (t["kind"].as_str().unwrap(),v){
   ("unit",V::Unit)=>{},("bool",V::Bool(b))=>w.bytes(&[u8::from(*b)])?,
   ("u32",V::U32(n))=>w.bytes(&n.to_le_bytes())?,("u64",V::U64(n))=>w.bytes(&n.to_le_bytes())?,
   ("i32",V::I32(n))=>w.bytes(&n.to_le_bytes())?,("f32",V::F32(n))=>w.bytes(&n.to_le_bytes())?,
   ("int",V::Int(n))=>w.bytes(n.to_string().as_bytes())?,
   ("string",V::String(n))|("enum",V::Enum(n))=>w.bytes(n.as_bytes())?,
   ("id",V::Id(_,n))=>w.bytes(format!("{n:032x}").as_bytes())?,
   ("record",V::Record(_,fs))=>{for f in t["fields"].as_array().unwrap(){binary_dynamic_write(s,&f["type"],fs.get(f["name"].as_str().unwrap()).ok_or("binary result field")?,w)?;}},
   ("list",V::List(xs))|("vector",V::Vector(xs))=>{w.word(xs.len().try_into().map_err(|_|"binary sequence size")?)?;for x in xs{binary_dynamic_write(s,&t["element"],x,w)?;}},
   ("option",V::Option(None))=>w.bytes(&[0])?,
   ("option",V::Option(Some(x)))=>{w.bytes(&[1])?;binary_dynamic_write(s,&t["element"],x,w)?;},
   ("result",V::Result(ok,x))=>{w.bytes(&[u8::from(!ok)])?;binary_dynamic_write(s,if *ok{&t["ok"]}else{&t["error"]},x,w)?;},
   _=>return Err("binary result type mismatch".into()),
  }Ok(())
 })
}
impl State{
 fn binary_pure_index(&mut self,index:usize,captured:Vec<V>)->R<V>{
  self.ctx.values.clear();self.ctx.arena_bytes=0;self.ctx.error=None;self.ctx.fuel=100000000;self.ctx.staged.clear();
  let handles=captured.into_iter().map(|v|self.ctx.put(v)).collect::<Vec<_>>();
  let result=(||{
   if self.ctx.error.is_some(){return Err("Ink value arena limit exceeded".into());}
   let h=unsafe{ink_program_call((&mut *self.ctx as *mut InkContext).cast(),0,index as u32,handles.as_ptr(),handles.len())};
   if let Some(e)=self.ctx.error.take(){return Err(match e{Failure::Host(e)=>e,Failure::Abort(_)=>"unhandled domain error".into()});}
   self.ctx.value(h)
  })();
  self.ctx.values.clear();self.ctx.arena_bytes=0;result
 }
 fn binary_call_frame(&mut self,name:&str,record:value_wire::Validated<'_>,output:&value_wire::Schema,limits:ValueWireLimits)->R<Vec<u8>>{
  let declarations=self.ctx.meta["functions"].as_array().ok_or("compiled functions")?;
  let(index,d)=declarations.iter().enumerate().find(|(_,d)|d["name"]==name).ok_or("unknown binary function")?;
  if d["internal"].as_bool()==Some(true){return Err("temporary storage is internal to a pure invocation".into());}
  let guard=d["precondition"].as_str().map(|name|declarations.iter().position(|d|d["name"]==name).ok_or("compiled precondition")).transpose()?;
  let captured=record.fields()?.map(|x|binary_dynamic_value(x?)).collect::<R<Vec<_>>>()?;
  if let Some(guard)=guard{if self.binary_pure_index(guard,captured.clone())?!=V::Bool(true){return Err("contract precondition failed".into());}}
  let result=self.binary_pure_index(index,captured)?;
  output.encode_with(limits,|w|binary_dynamic_write(output,output.root(),&result,w))
 }
}
